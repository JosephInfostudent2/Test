Further Analysis

yearly_library_df <- library_df |>
  group_by(checkoutyear) |>
  summarize(
    total_checkouts = sum(total_checkouts)
    
  )

# Seattle population data
seattle_population <- population_df |>
  filter(JURISDICTION == "Seattle", FILTER == 4)

seattle_population <- seattle_population |>
  pivot_longer( #changing cols to rows
    cols = starts_with("POP_"), #find cols that start with "POP_"
    names_to = "year", #change the name of those cols to year and store the previous cols as rows
    values_to = "population" #create another col to store previous values as rows
  ) |>
  mutate(
    year = as.numeric(str_remove(year, "POP_")) #remove the prefix, typecast it to integer and update year col
  ) |>
  select(year, population)

seattle_public_library_and_population <- left_join(yearly_library_df, seattle_population, join_by("checkoutyear" == "year"))
